第 1 步：在微信公众平台配置服务器地址 (URL)
登录微信公众平台（mp.weixin.qq.com），进入「设置与开发」-「基本配置」-「服务器配置」，启用并填写以下信息：
服务器地址 (URL)
其中 {accountKey} 替换为您即将登记的服务号标识（例如 mp_official）。
令牌 (Token)
自定义字符串（3-32位），需与网关中录入的 Token 保持完全一致。
消息加解密密钥
由微信随机生成 43 位 EncodingAESKey，需在网关登记用于解密密文事件。
加解密方式
安全模式 (推荐) 或 兼容模式。网关会自动解析密文并转发明文 JSON。
第 2 步：在网关控制台登记微信服务号
网关支持多租户/多服务号同时运行。每个服务号根据其标识 Key 进行独立的 URL 路由、URL 验证（echostr）与 AES 密文解密。
  1. 切换至左侧菜单 「微信服务号」 页面；
  2. 点击右上角 「新增服务号」 按钮；
  3. 填写 服务号名称（如：主官方商城服务号）与 标识 Key（如：shop_mp）；
  4. 填入服务号的 AppID、Token 以及 EncodingAESKey；
  5. 保存后，微信后台服务器 URL 填入 https://your-domain.com/wechat/shop_mp 即可一键通过验证保存。
第 3 步：配置下游业务系统订阅方与过滤规则
一个服务号的消息可扇出（Fan-out）广播投递给多个下游业务系统（CRM、客服系统、订单系统、数据分析平台等）。各下游通过独立的过滤规则订阅所需事件。
目标 Webhook
下游系统接收 POST 请求的 HTTP/HTTPS 接口地址。支持超时设置（默认 5s）与最大重试次数（默认 3 次）。
事件过滤规则
支持根据微信消息属性精准过滤。留空则订阅该服务号所有事件。
过滤规则 JSON 格式示例：
// 示例 1：仅接收关注与取关事件
{"msg_type": "event", "event": "subscribe"}

// 示例 2：仅接收模板消息发送结果回调
{"msg_type": "event", "event": "TEMPLATESENDJOBFINISH"}

// 示例 3：仅接收用户发送的文本上行消息
{"msg_type": "text"}
第 4 步：下游系统集成验签与响应契约
网关向订阅方发送 POST 请求，携带统一解析后的 JSON 负载与安全防篡改签名 Header。下游系统校验签名后返回 HTTP 200 状态码即可。
请求头 (Headers)
X-Event-Id: <evt_xxx>
X-Event-Type: <event_type>
X-Timestamp: <unix_seconds>
X-Event-Signature: sha256=<hmac_sha256_hex_digest>
签名计算规则
HMAC-SHA256(secret_key, raw_http_body)，计算结果十六进制编码。前缀包含 sha256=。
响应契约
下游必须在限定时间内返回 HTTP 状态码 200 或 204。若返回非 2xx 或发生超时，网关将按照指数退避策略自动重试投递。 
import hmac
import hashlib
from fastapi import FastAPI, Request, HTTPException

app = FastAPI()
WEBHOOK_SECRET = "your_subscriber_secret"

@app.post("/webhook/wechat")
async def handle_wechat_event(request: Request):
    signature_header = request.headers.get("X-Event-Signature", "")
    body_bytes = await request.body()

    # 1. 验证 HMAC-SHA256 签名防伪造
    expected_sig = "sha256=" + hmac.new(
        WEBHOOK_SECRET.encode(),
        body_bytes,
        hashlib.sha256
    ).hexdigest()

    if not hmac.compare_digest(signature_header, expected_sig):
        raise HTTPException(status_code=401, detail="Invalid signature")

    # 2. 解析网关标准化 JSON 结构
    payload = await request.json()
    event_id = request.headers.get("X-Event-Id")
    msg_type = payload.get("msg_type")
    print(f"收到事件 [{event_id}], 类型: {msg_type}")

    # 3. 业务处理完毕，返回 200 确认投递成功
    return {"status": "ok"}
package main

import (
    "crypto/hmac"
    "crypto/sha256"
    "encoding/hex"
    "io"
    "net/http"
)

const webhookSecret = "your_subscriber_secret"

func webhookHandler(w http.ResponseWriter, r *http.Request) {
    body, err := io.ReadAll(r.Body)
    if err != nil {
        http.Error(w, "Read error", http.StatusBadRequest)
        return
    }

    // 1. 验证 HMAC-SHA256 签名
    mac := hmac.New(sha256.New, []byte(webhookSecret))
    mac.Write(body)
    expectedSig := "sha256=" + hex.EncodeToString(mac.Sum(nil))

    if !hmac.Equal([]byte(r.Header.Get("X-Event-Signature")), []byte(expectedSig)) {
        http.Error(w, "Unauthorized", http.StatusUnauthorized)
        return
    }

    // 2. 业务处理（如写入消息队列或更新数据库）...

    // 3. 响应 200 确认消费完成
    w.WriteHeader(http.StatusOK)
    w.Write([]byte(`{"status":"ok"}`))
}
const express = require('express');
const crypto = require('crypto');

const app = express();
const WEBHOOK_SECRET = 'your_subscriber_secret';

// 注意：需获取原始 raw body 参与计算签名
app.post('/webhook/wechat', express.raw({ type: '*/*' }), (req, res) => {
    const rawBody = req.body;
    const clientSig = req.headers['x-event-signature'] || '';

    // 1. 计算 HMAC-SHA256 签名
    const expectedSig = 'sha256=' + crypto
        .createHmac('sha256', WEBHOOK_SECRET)
        .update(rawBody)
        .digest('hex');

    if (clientSig !== expectedSig) {
        return res.status(401).send('Invalid signature');
    }

    // 2. 解析 JSON 负载
    const event = JSON.parse(rawBody.toString('utf8'));
    console.log('接收到合法事件:', event.event_id, event.msg_type);

    // 3. 响应 200
    res.status(200).json({ status: 'ok' });
});
